## 一、现状分析

### 1.1 当前架构概述

## 二、架构设计原则

### 2.1 核心原则

1. **MCP 原子化** - MCP 是"铲子"，做最底层的事
2. **Skill 场景化** - Skill 是"菜谱"，封装完整工作流
3. **配置驱动** - 工具/Skill 可通过配置声明，无需硬编码
4. **用户可扩展** - 支持用户在页面新建和管理自定义 Skills
5. **热加载** - 配置变更无需重启服务

### 3.3 工具注册机制

```python
class ToolRegistry:
    """工具注册中心 - 支持动态注册和热加载"""
    
    def __init__(self):
        self._tools: Dict[str, ToolDefinition] = {}
    
    def register(self, tool: ToolDefinition):
        """注册工具"""
        self._tools[tool.name] = tool
        logger.info(f"[ToolRegistry] 注册工具: {tool.name}")
    
    def register_from_config(self, config: List[dict]):
        """从配置注册工具"""
        for tool_config in config:
            tool = ToolDefinition(**tool_config)
            self.register(tool)
    
    def get_tool(self, name: str) -> Optional[ToolDefinition]:
        """获取工具定义"""
        return self._tools.get(name)
    
    def list_tools(self, category: str = None) -> List[ToolDefinition]:
        """列出工具"""
        tools = self._tools.values()
        if category:
            tools = [t for t in tools if t.category == category]
        return tools
    
    def reload(self):
        """热重载 - 从数据库/配置文件重新加载"""
        # 从数据库加载启用的工具
        tools = db.get_enabled_tools()
        self._tools.clear()
        for tool_config in tools:
            self.register(ToolDefinition(**tool_config))
```

### 3.4 核心工具实现

#### 3.4.1 query_data 工具

```python
@tool(
    name="query_data",
    description="使用自然语言查询数据，系统会自动生成 SQL 并执行",
    input_schema={
        "query": str,
        "limit": int
    }
)
async def query_data(args: dict) -> dict:
    """通用数据查询工具
    
    用户可以用自然语言描述想要查询的数据，系统：
    1. 理解用户意图
    2. 映射到数据表和字段
    3. 生成 SQL
    4. 执行并返回结果
    """
    user_query = args.get("query")
    limit = args.get("limit", 100)
    
    # 1. 获取可用表结构
    tables = get_available_tables()
    
    # 2. 构建上下文给 LLM
    context = build_query_context(tables)
    
    # 3. 让 LLM 生成 SQL
    sql = await generate_sql(user_query, context)
    
    # 4. 验证 SQL 安全性
    if not is_safe_sql(sql):
        return {"error": "查询包含不允许的操作"}
    
    # 5. 执行查询
    result = execute_query(sql, limit)
    
    return {
        "data": result,
        "sql": sql,
        "count": len(result)
    }
```

#### 3.4.2 generate_excel 工具

```python
@tool(
    name="generate_excel",
    description="根据数据和模板生成 Excel 文件",
    input_schema={
        "data": dict,
        "template": str,
        "filename": str
    }
)
async def generate_excel(args: dict) -> dict:
    """Excel 生成工具
    
    支持：
    - 传入数据字典，直接写入 Excel
    - 指定模板，使用模板样式
    - 自动列宽调整
    - 多 Sheet 支持
    """
    data = args.get("data", {})
    template = args.get("template")
    filename = args.get("filename", "export.xlsx")
    
    # 创建工作簿
    wb = Workbook()
    ws = wb.active
    ws.title = "数据"
    
    # 写入数据
    if isinstance(data, list):
        # 列表数据，写入多行
        for row_idx, row in enumerate(data, 1):
            if isinstance(row, dict):
                for col_idx, (key, value) in enumerate(row.items(), 1):
                    ws.cell(row=row_idx, column=col_idx, value=value)
            else:
                ws.cell(row=row_idx, column=1, value=row)
    elif isinstance(data, dict):
        # 字典数据，key-value 格式
        for row_idx, (key, value) in enumerate(data.items(), 1):
            ws.cell(row=row_idx, column=1, value=key)
            ws.cell(row=row_idx, column=2, value=str(value))
    
    # 保存文件
    output_path = f"/tmp/{filename}"
    wb.save(output_path)
    
    return {
        "success": True,
        "file_path": output_path,
        "filename": filename
    }
```

```python
@tool(
    name="parse_attachment",
    description="通用附件解析，自动识别图片/表格/文档并提取内容",
    input_schema={"file_path": str, "options": dict}
)
async def parse_attachment(args: dict) -> dict:
    """支持图片、PDF、Excel、Word 等多种格式"""
    file_path = args.get("file_path")
    file_type = detect_file_type(file_path)  # 自动识别
    
    if file_type in ["jpg", "png", "gif", "bmp", "webp"]:
        result = await parse_image(file_path)  # OCR + 场景理解
    elif file_type == "pdf":
        result = await parse_pdf(file_path)     # 文本 + 表格提取
    elif file_type in ["xlsx", "xls", "csv"]:
        result = await parse_excel(file_path)   # 结构化数据
    elif file_type in ["doc", "docx"]:
        result = await parse_word(file_path)    # 文本提取
    
    return {"file_type": file_type, "content": result.get("content"),
            "tables": result.get("tables"), "summary": result.get("summary")}
```

### 4.4 Skill 编排器

```python
class SkillOrchestrator:
    """Skill 编排器 - 支持串行/并行/条件分支"""
    
    async def execute(
        self, 
        skill: SkillDefinition, 
        context: dict
    ) -> dict:
        """执行 Skill"""
        
        if skill.workflow_type == "linear":
            return await self._execute_linear(skill, context)
        elif skill.workflow_type == "parallel":
            return await self._execute_parallel(skill, context)
        elif skill.workflow_type == "conditional":
            return await self._execute_conditional(skill, context)
    
    async def _execute_linear(
        self, 
        skill: SkillDefinition, 
        context: dict
    ) -> dict:
        """串行执行"""
        results = []
        current_context = context
        
        for step in skill.workflow_config["steps"]:
            # 调用 MCP 工具
            tool_name = step["tool"]
            tool_input = self._build_tool_input(step, current_context)
            
            result = await mcp_server.call_tool(tool_name, tool_input)
            results.append({
                "step": step["name"],
                "result": result
            })
            
            # 更新上下文
            current_context.update(result)
        
        return {"results": results}
    
    async def _execute_parallel(
        self, 
        skill: SkillDefinition, 
        context: dict
    ) -> dict:
        """并行执行"""
        tasks = []
        for step in skill.workflow_config["steps"]:
            tool_name = step["tool"]
            tool_input = self._build_tool_input(step, context)
            tasks.append(mcp_server.call_tool(tool_name, tool_input))
        
        results = await asyncio.gather(*tasks)
        
        return {"results": results}
```

### 6.2 动态工具描述生成

```python
def build_tools_description(tools: List[ToolDefinition]) -> str:
    """动态构建工具描述"""
    
    # 按类别分组
    categories = {}
    for tool in tools:
        if tool.category not in categories:
            categories[tool.category] = []
        categories[tool.category].append(tool)
    
    # 构建描述
    lines = []
    for category, tool_list in categories.items():
        lines.append(f"\n### {category}")
        for tool in tool_list:
            params = ", ".join(tool.input_schema.get("required", []))
            lines.append(f"- **{tool.name}**: {tool.description} (参数: {params})")
    
    return "\n".join(lines)
```

```python
class PluginLoader:
    """插件加载器 - 支持热加载"""
    
    def load_plugins(self, plugin_dir: str):
        """加载所有插件"""
        for item in Path(plugin_dir).iterdir():
            if item.is_dir() and (item / "plugin.json").exists():
                self.load_plugin(item)
    
    def reload_plugin(self, plugin_name: str):
        """热重载单个插件"""
        # 1. 卸载旧插件
        self._unload_plugin(plugin_name)
        
        # 2. 重新加载
        plugin_path = Path(plugin_dir) / plugin_name
        self.load_plugin(plugin_path)
        
        # 3. 通知 Agent 刷新工具列表
        event_bus.publish("plugins.reloaded", plugin_name)
```
